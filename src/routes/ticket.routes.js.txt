import { Router } from 'express';
import { authenticateToken } from '../middleware/auth.middleware.js';
import { getSupabaseClient } from '../config/supabase.js';
import { analyzeCustomerIssue } from '../services/ai.service.js';
import { ticketSchema } from '../validators/ticket.validator.js';

const router = Router();

router.post('/', authenticateToken, async (req, res, next) => {
  try {
    const parsed = ticketSchema.safeParse(req.body);

    if (!parsed.success) {
      return res.status(400).json({
        success: false,
        error: 'Invalid ticket data.',
        details: parsed.error.flatten(),
      });
    }

    const { sector, issueDescription } = parsed.data;

    // AI analysis
    const aiResult = await analyzeCustomerIssue(
      sector,
      issueDescription
    );

    const supabase = getSupabaseClient();

    const { data: ticket, error } = await supabase
      .from('tickets')
      .insert({
        customer_id: req.user.id,
        sector,
        issue_description: issueDescription,
        ai_summary: aiResult.summary,
        ai_category: aiResult.category,
        ai_priority: aiResult.priority,
        ai_sentiment: aiResult.sentiment,
        ai_recommended_action: aiResult.recommendedAction,
      })
      .select()
      .single();

    if (error) {
      return next(error);
    }

    res.status(201).json({
      success: true,
      message: 'Ticket created successfully.',
      ticket,
    });

  } catch (error) {
    next(error);
  }
});

export default router;
